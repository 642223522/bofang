// WidgetCrypto.js.txt — Widget.crypto JS 胶水（run #35 第二轮，二进制桥）
//
// 组装 Widget.crypto.* 友好 API。原生原语（__fwCryptoAESEncrypt / __fwCryptoAESDecrypt /
// __fwCryptoXOR / __fwCryptoHash / __fwCryptoHMAC / __fwCryptoRandomBytes）由
// Core/WidgetCrypto.swift 挂到 JSContext 全局；输入输出全是 base64 字符串。
// 本文件只做：按字段编码转换、校验、错误映射、GCM tag/aad 组装。
//
// 编码约定（三家 AI 评审定稿）：
// - 不自动识别编码。每个字节参数的编码按字段显式声明：inputEncoding（数据/密文）、
//   keyEncoding、ivEncoding、outputEncoding，GCM 加 tagEncoding/aadEncoding。
// - 默认值：解密 in=base64、key/iv=utf8、out=utf8；加密 out=base64。
//   （全局二进制默认是 base64；key/iv 默认 utf8，因为视频站的 key/iv 常是可读字符串。）
// - 编码名大小写不敏感，"utf-8" 归一为 "utf8"；支持 utf8 / base64 / base64url / hex。
// - 解密结果不是合法 UTF-8 时抛错（不静默替换字符）；outputEncoding 可取 base64/hex 拿原始字节。
//
// 错误约定：
// - 参数/契约错 → TypeError（key/iv/tag 长度、未知 mode/padding/编码、ECB 带 iv、
//   CTR/GCM 传 pkcs7/zero、非法 hex/base64）。
// - 运行时失败 → Error，带稳定 .code（CRYPTO_BAD_PADDING / CRYPTO_ALIGNMENT_ERROR /
//   CRYPTO_AUTH_FAILED / CRYPTO_INVALID_UTF8 / CRYPTO_OS_ERROR）。
// - 超限（输入 > 5MB、randomBytes n 越界）→ RangeError。
// - 绝不静默返回 null/空串/原样数据；错误信息只写长度和期望长度，不回显 key/iv 内容。
//
// GCM 约定：
// - tag 固定 16 字节。加密默认把 tag 拼在密文末尾（返回 base64(ciphertext + tag)）；
//   解密默认从输入末尾 16 字节拆 tag；options.tag 单独传时按单独的 tag 用，数据即纯密文。
// - aad 经 options.aad + options.aadEncoding 传入（默认 utf8，可为空）。
// - nonce（iv）12 或 16 字节；加密时 iv 必传（不自动生成——返回值是单个字符串，
//   装不下自动生成的 nonce；要随机 nonce 先调 randomBytes(12) 再传进来）。
//
// 注意：本文件是 .txt 资源（Bundle 读盘后 evaluateScript），不是 Swift """ 字符串，
// 所以反斜杠可用；但 harness 把它嵌进 JS 模板字面量，故不用反引号与 ${。

(function (g) {
    'use strict';

    // 把 6 个原生原语收进闭包局部变量（胶水加载后 Swift 侧会删掉全局上的 __fwCrypto*，
    // 模块就调不到原语、绕不过胶水的校验；胶水内部用这里的局部变量继续调原语）。
    var _aesEncrypt = g.__fwCryptoAESEncrypt;
    var _aesDecrypt = g.__fwCryptoAESDecrypt;
    var _xor = g.__fwCryptoXOR;
    var _hash = g.__fwCryptoHash;
    var _hmac = g.__fwCryptoHMAC;
    var _randomBytes = g.__fwCryptoRandomBytes;

    // ---------- 基础工具 ----------

    var MAX_INPUT_BYTES = 5 * 1024 * 1024; // 单次输入总长度护栏，按解码后的字节数计（与 Swift 侧 maxInputBytes 同口径）。
    var MAX_RANDOM_BYTES = 65536;

    function isObject(v) {
        return v !== null && typeof v === 'object' && !Array.isArray(v);
    }

    function normEncoding(enc, def) {
        if (enc === undefined || enc === null) { return def; }
        var e = String(enc).toLowerCase().replace(/-/g, '');
        if (e === 'utf8' || e === 'base64' || e === 'base64url' || e === 'hex') { return e; }
        throw new TypeError('unknown encoding: ' + String(enc));
    }

    // ---------- UTF-8 编解码（裸 JSContext 没有 TextEncoder/TextDecoder，手工实现） ----------

    // UTF-16 字符串 → UTF-8 字节数组（处理代理对）。
    function utf8Encode(str) {
        var bytes = [];
        for (var i = 0; i < str.length; i++) {
            var cp = str.charCodeAt(i);
            // 孤立代理（高代理后不跟低代理，或单独的低代理）是无效 UTF-16，必须抛错，
            // 不能悄悄编成 3 字节的 CESU-8（那是无效 UTF-8，与 utf8Decode 的拦截不一致）。
            if (cp >= 0xD800 && cp <= 0xDBFF) {
                if (i + 1 < str.length) {
                    var lo = str.charCodeAt(i + 1);
                    if (lo >= 0xDC00 && lo <= 0xDFFF) {
                        cp = 0x10000 + ((cp - 0xD800) << 10) + (lo - 0xDC00);
                        i++;
                    } else {
                        throw new TypeError('isolated surrogate in input string');
                    }
                } else {
                    throw new TypeError('isolated surrogate in input string');
                }
            } else if (cp >= 0xDC00 && cp <= 0xDFFF) {
                throw new TypeError('isolated surrogate in input string');
            }
            if (cp < 0x80) {
                bytes.push(cp);
            } else if (cp < 0x800) {
                bytes.push(0xC0 | (cp >> 6), 0x80 | (cp & 0x3F));
            } else if (cp < 0x10000) {
                bytes.push(0xE0 | (cp >> 12), 0x80 | ((cp >> 6) & 0x3F), 0x80 | (cp & 0x3F));
            } else {
                bytes.push(0xF0 | (cp >> 18), 0x80 | ((cp >> 12) & 0x3F), 0x80 | ((cp >> 6) & 0x3F), 0x80 | (cp & 0x3F));
            }
        }
        return bytes;
    }

    // UTF-8 字节数组 → UTF-16 字符串。严格解码：非法序列（超长编码、代理区码点、
    // 超出 U+10FFFF、截断）抛 CRYPTO_INVALID_UTF8，不静默替换字符。
    function utf8Decode(bytes) {
        var out = '';
        var i = 0;
        while (i < bytes.length) {
            var b0 = bytes[i];
            var cp, need;
            if (b0 < 0x80) {
                cp = b0; need = 0;
            } else if ((b0 & 0xE0) === 0xC0) {
                cp = b0 & 0x1F; need = 1;
                if (cp < 2) { throw invalidUtf8(); }
            } else if ((b0 & 0xF0) === 0xE0) {
                cp = b0 & 0x0F; need = 2;
            } else if ((b0 & 0xF8) === 0xF0) {
                cp = b0 & 0x07; need = 3;
                if (cp > 4) { throw invalidUtf8(); }
            } else {
                throw invalidUtf8();
            }
            if (i + need >= bytes.length) { throw invalidUtf8(); }
            for (var j = 1; j <= need; j++) {
                var bj = bytes[i + j];
                if ((bj & 0xC0) !== 0x80) { throw invalidUtf8(); }
                cp = (cp << 6) | (bj & 0x3F);
            }
            // 拒绝超长编码与非法码点。
            if ((need === 2 && cp < 0x800) || (need === 3 && cp < 0x10000)) { throw invalidUtf8(); }
            if (cp >= 0xD800 && cp <= 0xDFFF) { throw invalidUtf8(); }
            if (cp > 0x10FFFF) { throw invalidUtf8(); }
            if (cp < 0x10000) {
                out += String.fromCharCode(cp);
            } else {
                cp -= 0x10000;
                out += String.fromCharCode(0xD800 + (cp >> 10), 0xDC00 + (cp & 0x3FF));
            }
            i += need + 1;
        }
        return out;
    }

    function invalidUtf8() {
        var e = new Error('result is not valid UTF-8');
        e.code = 'CRYPTO_INVALID_UTF8';
        return e;
    }

    // ---------- base64 / hex ----------

    var B64_ALPHABET = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/';
    var B64_REV = {};
    for (var k = 0; k < 64; k++) { B64_REV[B64_ALPHABET.charAt(k)] = k; }

    // 字节数组 → base64（标准，带 = 填充）。
    function bytesToB64(bytes) {
        var out = '';
        for (var i = 0; i < bytes.length; i += 3) {
            var b0 = bytes[i], b1 = i + 1 < bytes.length ? bytes[i + 1] : 0, b2 = i + 2 < bytes.length ? bytes[i + 2] : 0;
            var n = (b0 << 16) | (b1 << 8) | b2;
            out += B64_ALPHABET.charAt((n >> 18) & 63);
            out += B64_ALPHABET.charAt((n >> 12) & 63);
            out += i + 1 < bytes.length ? B64_ALPHABET.charAt((n >> 6) & 63) : '=';
            out += i + 2 < bytes.length ? B64_ALPHABET.charAt(n & 63) : '=';
        }
        return out;
    }

    // base64/base64url 字符串 → 字节数组。容忍缺 = 填充与空白；非法字符抛 TypeError。
    function b64ToBytes(s, urlSafe) {
        var t = String(s).replace(/[\t\n\f\r ]/g, '');
        if (urlSafe) { t = t.replace(/-/g, '+').replace(/_/g, '/'); }
        if (/[^A-Za-z0-9+/=]/.test(t)) { throw new TypeError('invalid base64 input'); }
        // '=' 只能是末尾的填充（最多 2 个）。中间出现（"AB=C"、"AB=A"）是非法输入，
        // 必须抛错，不能多解出字节。
        var stripped = t.replace(/=+$/, '');
        if (stripped.indexOf('=') !== -1) { throw new TypeError('invalid base64 input'); }
        var padLen = t.length - stripped.length;
        if (padLen > 2) { throw new TypeError('invalid base64 input'); }
        // 带 = 填充时总长度必须是 4 的倍数："AB=" 是残缺填充，不能静默补成 "AB==" 多解出字节。
        if (padLen > 0 && t.length % 4 !== 0) { throw new TypeError('invalid base64 input'); }
        var rem = t.length % 4;
        if (rem === 1) { throw new TypeError('invalid base64 input'); }
        if (rem !== 0) { t += '==='.slice(0, 4 - rem); }
        var bytes = [];
        for (var i = 0; i < t.length; i += 4) {
            var c0 = B64_REV[t.charAt(i)], c1 = B64_REV[t.charAt(i + 1)];
            var c2 = t.charAt(i + 2) === '=' ? 0 : B64_REV[t.charAt(i + 2)];
            var c3 = t.charAt(i + 3) === '=' ? 0 : B64_REV[t.charAt(i + 3)];
            if (c0 === undefined || c1 === undefined || c2 === undefined || c3 === undefined) {
                throw new TypeError('invalid base64 input');
            }
            var n = (c0 << 18) | (c1 << 12) | (c2 << 6) | c3;
            bytes.push((n >> 16) & 0xFF);
            if (t.charAt(i + 2) !== '=') { bytes.push((n >> 8) & 0xFF); }
            if (t.charAt(i + 3) !== '=') { bytes.push(n & 0xFF); }
        }
        return bytes;
    }

    // 字节数组 → base64url（无 = 填充）。
    function bytesToB64Url(bytes) {
        return bytesToB64(bytes).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
    }

    // hex 字符串 → 字节数组。容忍 0x 前缀；奇数长度或非法字符抛 TypeError。
    function hexToBytes(s) {
        var t = String(s);
        if (t.slice(0, 2) === '0x' || t.slice(0, 2) === '0X') { t = t.slice(2); }
        if (t.length % 2 !== 0) { throw new TypeError('invalid hex input: odd length'); }
        if (/[^0-9a-fA-F]/.test(t)) { throw new TypeError('invalid hex input'); }
        var bytes = [];
        for (var i = 0; i < t.length; i += 2) {
            bytes.push(parseInt(t.slice(i, i + 2), 16));
        }
        return bytes;
    }

    // 字节数组 → hex（小写）。
    function bytesToHex(bytes) {
        var out = '';
        for (var i = 0; i < bytes.length; i++) {
            var h = bytes[i].toString(16);
            out += h.length === 1 ? '0' + h : h;
        }
        return out;
    }

    // ---------- 编码转换（字符串 → base64 / base64 → 字符串） ----------

    // 任意编码的字符串 → base64（给 Swift 原语用）。
    function strToB64(s, enc) {
        if (typeof s !== 'string') { throw new TypeError('expected a string'); }
        // 前置守卫：先按字符串长度拦超限，再解码。原来先把整串全量解成 JS 数组
        // 再 checkMaxBytes，30MB 输入会先 OOM 才抛 RangeError。
        // （每 UTF-16 单元至少编 1 个 UTF-8 字节，故 s.length > 5MB 时解码必超限。）
        if (enc === 'utf8') {
            if (s.length > MAX_INPUT_BYTES) { throw new RangeError('input exceeds ' + MAX_INPUT_BYTES + ' bytes'); }
        } else if (enc === 'hex') {
            var hx = s.slice(0, 2) === '0x' || s.slice(0, 2) === '0X' ? s.slice(2) : s;
            if (hx.length > MAX_INPUT_BYTES * 2) { throw new RangeError('input exceeds ' + MAX_INPUT_BYTES + ' bytes'); }
        } else if (enc === 'base64' || enc === 'base64url') {
            var bx = s.replace(/[\t\n\f\r ]/g, '');
            if (bx.length > Math.ceil(MAX_INPUT_BYTES / 3) * 4) { throw new RangeError('input exceeds ' + MAX_INPUT_BYTES + ' bytes'); }
        }
        switch (enc) {
            case 'utf8': return bytesToB64(utf8Encode(s));
            case 'base64': return bytesToB64(b64ToBytes(s, false));
            case 'base64url': return bytesToB64(b64ToBytes(s, true));
            case 'hex': return bytesToB64(hexToBytes(s));
            default: throw new TypeError('unknown encoding: ' + enc);
        }
    }

    // base64 → 目标编码的字符串。
    function b64ToStr(b64, enc) {
        var bytes = b64ToBytes(b64, false);
        switch (enc) {
            case 'utf8': return utf8Decode(bytes);
            case 'base64': return b64;
            case 'base64url': return bytesToB64Url(bytes);
            case 'hex': return bytesToHex(bytes);
            default: throw new TypeError('unknown encoding: ' + enc);
        }
    }

    // base64 → 字节数组（供长度校验用）。
    // O(1) 长度公式，不解码：输入是 strToB64 产出的已规范化标准 base64
    // （长度必为 4 的倍数、带正确 = 填充），字节数 = 字符数 * 3/4 - 末尾 = 个数。
    // 原来每次都全量解码成几百万元素的 JS 数组，5MB 输入反复解 4～6 次，极易触发真机内存告警。
    function b64ByteLength(b64) {
        var t = String(b64).replace(/[\t\n\f\r ]/g, '');
        var pad = 0;
        if (t.charAt(t.length - 1) === '=') { pad++; }
        if (t.charAt(t.length - 2) === '=') { pad++; }
        return Math.floor(t.length * 3 / 4) - pad;
    }

    // ---------- 错误映射 ----------

    // Swift 原语返回 {ok:false, code, message} 时，按 code 抛对应的 JS 错误。
    function throwNativeError(res) {
        var code = res.code || 'CRYPTO_OS_ERROR';
        var message = res.message || 'crypto operation failed';
        if (code === 'INVALID_PARAMETER') {
            throw new TypeError(message);
        }
        if (code === 'RANGE') {
            throw new RangeError(message);
        }
        var e = new Error(message);
        e.code = code;
        throw e;
    }

    function checkNative(res) {
        if (!res || res.ok !== true) {
            throwNativeError(res || { code: 'CRYPTO_OS_ERROR', message: 'crypto operation failed' });
        }
        return res;
    }

    // ---------- 校验 ----------

    function checkByteLength(b64, allowed, name) {
        var len = b64ByteLength(b64);
        if (allowed.indexOf(len) < 0) {
            throw new TypeError(name + ' must be ' + allowed.join('/') + ' bytes, got ' + len);
        }
        return len;
    }

    function checkMaxBytes(b64, name) {
        if (b64ByteLength(b64) > MAX_INPUT_BYTES) {
            throw new RangeError(name + ' exceeds ' + MAX_INPUT_BYTES + ' bytes');
        }
    }

    // ---------- AES ----------

    function normMode(mode) {
        if (mode === undefined || mode === null) { return 'CBC'; }
        var m = String(mode).toUpperCase();
        if (m === 'CBC' || m === 'ECB' || m === 'CTR' || m === 'GCM') { return m; }
        throw new TypeError('unknown mode: ' + String(mode));
    }

    function normPadding(padding, mode) {
        var p = padding === undefined || padding === null
            ? (mode === 'CBC' || mode === 'ECB' ? 'pkcs7' : 'none')
            : String(padding).toLowerCase();
        if (p !== 'pkcs7' && p !== 'none' && p !== 'zero') {
            throw new TypeError('unknown padding: ' + String(padding));
        }
        // CTR/GCM 不需要 padding，显式传 pkcs7/zero 直接抛错（不能含糊）。
        if ((mode === 'CTR' || mode === 'GCM') && p !== 'none') {
            throw new TypeError('padding must be none for ' + mode);
        }
        return p;
    }

    // 把 options 里的 iv/key/data 按字段编码转成 base64，并做长度校验。
    // 返回 {dataB64, keyB64, ivB64, tagB64, aadB64}（缺失的为 null）。
    function prepareAES(data, key, options, isEncrypt) {
        var opts = isObject(options) ? options : {};
        var mode = normMode(opts.mode);
        var padding = normPadding(opts.padding, mode);
        var inputEnc = normEncoding(opts.inputEncoding, isEncrypt ? 'utf8' : 'base64');
        var keyEnc = normEncoding(opts.keyEncoding, 'utf8');
        var ivEnc = normEncoding(opts.ivEncoding, 'utf8');
        var outputEnc = normEncoding(opts.outputEncoding, isEncrypt ? 'base64' : 'utf8');
        var tagEnc = normEncoding(opts.tagEncoding, 'base64');
        var aadEnc = normEncoding(opts.aadEncoding, 'utf8');

        if (typeof data !== 'string') { throw new TypeError('data must be a string'); }
        if (typeof key !== 'string') { throw new TypeError('key must be a string'); }
        var dataB64 = strToB64(data, inputEnc);
        var keyB64 = strToB64(key, keyEnc);
        checkMaxBytes(dataB64, 'data');
        checkByteLength(keyB64, [16, 24, 32], 'key');

        var ivB64 = null;
        if (mode === 'ECB') {
            // 先判 typeof 再判空串：String([]) === ''，原来先用 String(v) !== '' 判断，
            // 传 [] 会绕过类型校验、被静默当成“没传”。
            if (opts.iv !== undefined && opts.iv !== null) {
                if (typeof opts.iv !== 'string') { throw new TypeError('iv must be a string'); }
                if (opts.iv !== '') { throw new TypeError('ECB must not carry an iv'); }
            }
        } else {
            // CBC/CTR/GCM 的 iv 必传（加密不自动生成——返回值是单个字符串，装不下自动生成的 nonce）。
            if (opts.iv === undefined || opts.iv === null) {
                throw new TypeError('iv is required for ' + mode);
            }
            if (typeof opts.iv !== 'string') { throw new TypeError('iv must be a string'); }
            if (opts.iv === '') { throw new TypeError('iv is required for ' + mode); }
            ivB64 = strToB64(opts.iv, ivEnc);
            if (mode === 'GCM') {
                checkByteLength(ivB64, [12, 16], 'iv');
            } else {
                checkByteLength(ivB64, [16], 'iv');
            }
        }

        // 非 GCM 传 tag 是错：原来静默忽略，模块以为 tag 生效了、其实没传进去。
        if (mode !== 'GCM' && opts.tag !== undefined && opts.tag !== null) {
            throw new TypeError('tag is only valid for GCM');
        }
        var tagB64 = null;
        if (mode === 'GCM' && opts.tag !== undefined && opts.tag !== null) {
            if (typeof opts.tag !== 'string') { throw new TypeError('tag must be a string'); }
            // 显式传了空字符串 tag 是错（tag 必须是 16 字节），别当成“没传”去拆输入。
            if (String(opts.tag) === '') { throw new TypeError('tag must not be empty'); }
            tagB64 = strToB64(opts.tag, tagEnc);
            checkByteLength(tagB64, [16], 'tag');
        }

        var aadB64 = null;
        if (opts.aad !== undefined && opts.aad !== null) {
            if (typeof opts.aad !== 'string') { throw new TypeError('aad must be a string'); }
            // 非 GCM 传非空 aad 是错：原来先转码传给 Swift、再被静默丢弃。
            if (mode !== 'GCM' && opts.aad !== '') {
                throw new TypeError('aad is only valid for GCM');
            }
            if (opts.aad !== '') {
                aadB64 = strToB64(opts.aad, aadEnc);
                checkMaxBytes(aadB64, 'aad');
            }
        }

        // padding:'none' 的长度校验只在加密时由 JS 抛 TypeError；解密时交给 Swift，
        // 统一报 CRYPTO_ALIGNMENT_ERROR。原来解密也走 JS 的 TypeError，同一种错按
        // padding 分裂成两种错误类型（'none' → TypeError，'zero'/'pkcs7' → Error）。
        if (isEncrypt && (mode === 'CBC' || mode === 'ECB') && padding === 'none') {
            if (b64ByteLength(dataB64) % 16 !== 0) {
                throw new TypeError('data length must be a multiple of 16 with padding none');
            }
        }
        // padding:'zero' 的加密：JS 侧先把明文补零到 16 倍数，再按 none 调原语。
        if (isEncrypt && (mode === 'CBC' || mode === 'ECB') && padding === 'zero') {
            var bytes = b64ToBytes(dataB64, false);
            var rem = bytes.length % 16;
            if (rem !== 0) {
                for (var i = rem; i < 16; i++) { bytes.push(0); }
                dataB64 = bytesToB64(bytes);
            }
            padding = 'none';
        }

        return {
            dataB64: dataB64, keyB64: keyB64, ivB64: ivB64, tagB64: tagB64, aadB64: aadB64,
            mode: mode, padding: padding, outputEnc: outputEnc
        };
    }

    function aesDecrypt(data, key, options) {
        var p = prepareAES(data, key, options, false);
        // GCM 且没单独传 tag：默认从输入末尾 16 字节拆 tag。
        var tagB64 = p.tagB64;
        var dataB64 = p.dataB64;
        if (p.mode === 'GCM' && tagB64 === null) {
            var bytes = b64ToBytes(dataB64, false);
            if (bytes.length < 16) {
                throw new TypeError('GCM input must include a 16-byte tag');
            }
            tagB64 = bytesToB64(bytes.slice(bytes.length - 16));
            dataB64 = bytesToB64(bytes.slice(0, bytes.length - 16));
        }
        var res = checkNative(_aesDecrypt(dataB64, p.keyB64, p.ivB64, p.mode, p.padding, tagB64, p.aadB64));
        // padding:'zero' 的解密：去尾部 0x00（zero padding 的定义；不去的话尾部 \0
        // 会进 UTF-8 字符串，JSON.parse 直接抛错）。
        var ptB64 = res.plaintext;
        if (p.padding === 'zero') {
            var ptBytes = b64ToBytes(ptB64, false);
            var end = ptBytes.length;
            while (end > 0 && ptBytes[end - 1] === 0) { end--; }
            ptB64 = bytesToB64(ptBytes.slice(0, end));
        }
        return b64ToStr(ptB64, p.outputEnc);
    }

    function aesEncrypt(data, key, options) {
        var p = prepareAES(data, key, options, true);
        var res = checkNative(_aesEncrypt(p.dataB64, p.keyB64, p.ivB64, p.mode, p.padding, p.aadB64));
        // GCM：默认把 tag 拼在密文末尾（返回 base64(ciphertext + tag)）。
        var outB64 = res.ciphertext;
        if (p.mode === 'GCM' && res.tag) {
            var cb = b64ToBytes(outB64, false);
            var tb = b64ToBytes(res.tag, false);
            outB64 = bytesToB64(cb.concat(tb));
        }
        return b64ToStr(outB64, p.outputEnc);
    }

    // ---------- XOR ----------

    function xor(data, key, options) {
        var opts = isObject(options) ? options : {};
        var inputEnc = normEncoding(opts.inputEncoding, 'base64');
        var keyEnc = normEncoding(opts.keyEncoding, 'utf8');
        var outputEnc = normEncoding(opts.outputEncoding, 'base64');
        if (typeof data !== 'string') { throw new TypeError('data must be a string'); }
        if (typeof key !== 'string') { throw new TypeError('key must be a string'); }
        var dataB64 = strToB64(data, inputEnc);
        var keyB64 = strToB64(key, keyEnc);
        checkMaxBytes(dataB64, 'data');
        // key 也受 5MB 护栏（原来只拦空 key，不限长）。
        checkMaxBytes(keyB64, 'key');
        if (b64ByteLength(keyB64) === 0) { throw new TypeError('key must not be empty'); }
        var res = checkNative(_xor(dataB64, keyB64));
        return b64ToStr(res.result, outputEnc);
    }

    // ---------- base64 / hex ----------

    function base64Encode(data, options) {
        var opts = isObject(options) ? options : {};
        var inputEnc = normEncoding(opts.inputEncoding, 'utf8');
        if (typeof data !== 'string') { throw new TypeError('data must be a string'); }
        var b64 = strToB64(data, inputEnc);
        checkMaxBytes(b64, 'data');
        if (opts.urlSafe) { return bytesToB64Url(b64ToBytes(b64, false)); }
        if (opts.noPadding) { return b64.replace(/=+$/, ''); }
        return b64;
    }

    function base64Decode(data, options) {
        var opts = isObject(options) ? options : {};
        var outputEnc = normEncoding(opts.outputEncoding, 'utf8');
        if (typeof data !== 'string') { throw new TypeError('data must be a string'); }
        var urlSafe = !!opts.urlSafe;
        var b64 = bytesToB64(b64ToBytes(data, urlSafe));
        checkMaxBytes(b64, 'data');
        return b64ToStr(b64, outputEnc);
    }

    function hexEncode(data, options) {
        var opts = isObject(options) ? options : {};
        var inputEnc = normEncoding(opts.inputEncoding, 'utf8');
        if (typeof data !== 'string') { throw new TypeError('data must be a string'); }
        var b64 = strToB64(data, inputEnc);
        checkMaxBytes(b64, 'data');
        return bytesToHex(b64ToBytes(b64, false));
    }

    function hexDecode(data, options) {
        var opts = isObject(options) ? options : {};
        var outputEnc = normEncoding(opts.outputEncoding, 'utf8');
        if (typeof data !== 'string') { throw new TypeError('data must be a string'); }
        var b64 = bytesToB64(hexToBytes(data));
        checkMaxBytes(b64, 'data');
        return b64ToStr(b64, outputEnc);
    }

    // ---------- hash / hmac ----------

    function normHashAlgorithm(algo) {
        if (algo === undefined || algo === null) { return 'sha256'; }
        var a = String(algo).toLowerCase().replace(/-/g, '');
        if (a === 'md5' || a === 'sha1' || a === 'sha256' || a === 'sha512') { return a; }
        throw new TypeError('unknown hash algorithm: ' + String(algo));
    }

    function hash(data, options) {
        var opts = isObject(options) ? options : {};
        var algorithm = normHashAlgorithm(opts.algorithm);
        var inputEnc = normEncoding(opts.inputEncoding, 'utf8');
        var outputEnc = normEncoding(opts.outputEncoding, 'hex');
        // 摘要是任意字节，按严格 UTF-8 解几乎必抛 CRYPTO_INVALID_UTF8——utf8 输出不认。
        if (outputEnc === 'utf8') { throw new TypeError('outputEncoding must be hex, base64 or base64url for hash'); }
        if (typeof data !== 'string') { throw new TypeError('data must be a string'); }
        var dataB64 = strToB64(data, inputEnc);
        checkMaxBytes(dataB64, 'data');
        var res = checkNative(_hash(algorithm, dataB64));
        // 原语返回 hex；outputEncoding 为 hex 时直接返回，否则转码。
        if (outputEnc === 'hex') { return res.result; }
        return b64ToStr(bytesToB64(hexToBytes(res.result)), outputEnc);
    }

    function hmac(algorithm, data, key, options) {
        var opts = isObject(options) ? options : {};
        var algo = normHashAlgorithm(algorithm);
        var inputEnc = normEncoding(opts.inputEncoding, 'utf8');
        var keyEnc = normEncoding(opts.keyEncoding, 'utf8');
        var outputEnc = normEncoding(opts.outputEncoding, 'hex');
        if (outputEnc === 'utf8') { throw new TypeError('outputEncoding must be hex, base64 or base64url for hmac'); }
        if (typeof data !== 'string') { throw new TypeError('data must be a string'); }
        if (typeof key !== 'string') { throw new TypeError('key must be a string'); }
        var dataB64 = strToB64(data, inputEnc);
        var keyB64 = strToB64(key, keyEnc);
        checkMaxBytes(dataB64, 'data');
        checkMaxBytes(keyB64, 'key');
        var res = checkNative(_hmac(algo, dataB64, keyB64));
        if (outputEnc === 'hex') { return res.result; }
        return b64ToStr(bytesToB64(hexToBytes(res.result)), outputEnc);
    }

    // ---------- random ----------

    function randomBytes(n, options) {
        var opts = isObject(options) ? options : {};
        var outputEnc = normEncoding(opts.outputEncoding, 'base64');
        // 随机字节是任意字节，按严格 UTF-8 解几乎必抛 CRYPTO_INVALID_UTF8——utf8 输出不认。
        if (outputEnc === 'utf8') { throw new TypeError('outputEncoding must be hex, base64 or base64url for randomBytes'); }
        if (typeof n !== 'number' || Math.floor(n) !== n || n < 0 || n > MAX_RANDOM_BYTES) {
            throw new RangeError('n must be an integer 0...' + MAX_RANDOM_BYTES);
        }
        var res = checkNative(_randomBytes(n));
        return b64ToStr(res.result, outputEnc);
    }

    // ---------- 挂到 Widget.crypto ----------

    var crypto = {
        aesDecrypt: aesDecrypt,
        aesEncrypt: aesEncrypt,
        xor: xor,
        base64Encode: base64Encode,
        base64Decode: base64Decode,
        hexEncode: hexEncode,
        hexDecode: hexDecode,
        hash: hash,
        md5: function (data, options) {
            // 浅拷贝：别改写调用方传进来的 options 对象。
            var opts = {};
            if (isObject(options)) {
                for (var k in options) { if (Object.prototype.hasOwnProperty.call(options, k)) { opts[k] = options[k]; } }
            }
            opts.algorithm = 'md5';
            return hash(data, opts);
        },
        sha1: function (data, options) {
            // 浅拷贝：别改写调用方传进来的 options 对象。
            var opts = {};
            if (isObject(options)) {
                for (var k in options) { if (Object.prototype.hasOwnProperty.call(options, k)) { opts[k] = options[k]; } }
            }
            opts.algorithm = 'sha1';
            return hash(data, opts);
        },
        sha256: function (data, options) {
            // 浅拷贝：别改写调用方传进来的 options 对象。
            var opts = {};
            if (isObject(options)) {
                for (var k in options) { if (Object.prototype.hasOwnProperty.call(options, k)) { opts[k] = options[k]; } }
            }
            opts.algorithm = 'sha256';
            return hash(data, opts);
        },
        sha512: function (data, options) {
            // 浅拷贝：别改写调用方传进来的 options 对象。
            var opts = {};
            if (isObject(options)) {
                for (var k in options) { if (Object.prototype.hasOwnProperty.call(options, k)) { opts[k] = options[k]; } }
            }
            opts.algorithm = 'sha512';
            return hash(data, opts);
        },
        hmac: hmac,
        randomBytes: randomBytes
    };

    // 冻住 Widget.crypto 及其方法，防止模块互相篡改。
    Object.freeze(crypto);

    // Object.freeze(crypto) 只冻住 crypto 内部成员，g.Widget.crypto 本身仍可写可删，
    // 模块一句 Widget.crypto = fakeObj 就能整个替换掉。改用 defineProperty 锁定属性。
    // 已挂载过（同一 JSContext 重复 install）直接保留已有的：configurable:false 的属性
    // 再 defineProperty 会抛 TypeError；重复挂载的 crypto 与这份功能相同，不必重挂。
    if (!g.Widget) { g.Widget = {}; }
    if (!g.Widget.crypto) {
        Object.defineProperty(g.Widget, 'crypto', { value: crypto, writable: false, configurable: false, enumerable: true });
    }
})(this);
