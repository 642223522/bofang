// Widget.html — Forward 兼容的 Cheerio DOM 解析接口（宿主提供）。
//
// 用法（与 Forward 官方一致，先拿 http.get 的文本再解析）：
//   const res = await Widget.http.get(url);
//   const $ = Widget.html.load(res.data);
//   const titles = $('.card h2').map((i, el) => $(el).text()).get();
//
//   const $2 = await Widget.html.loadURL(url);   // 我们自己的扩展：get + load 一步到位
//
// 底层是 cheerio 1.2.0（parse5 + htmlparser2 + css-select + 完整实体表），
// 由 esbuild 打成单文件（Core/cheerio.min.js.txt，.txt 资源进包，不违反 0 .js 门禁），
// 运行时由 Swift 侧 __fwEnsureCheerio 懒注入（第一次调 load/loadURL 时才注入，
// 约 30ms，之后常驻；不用的模块不付这个钱）。
// 手写解析器已验证走不通（table foster parenting、raw-text 边界、实体解码陷阱），
// 不要改回手写；想升级 cheerio 版本先跑 round30-pack 的沙箱 harness。
//
// 护栏（load 是同步纯解析，解析本身不触网）：
// - html 必须是字符串，非字符串抛 TypeError；空串返回空文档（不抛）；
// - 输入按 UTF-8 字节数限 5MB（HARD_MAX_HTML_BYTES 是硬上限，只许调低不许调高；
//   maxBytes/maxHTMLBytes 任一有效（>= 1 的有限数），无效（非数/NaN/Infinity/< 1）
//   抛 TypeError，null 视同未传；maxBytes 优先于 maxHTMLBytes）；
//   注意：5MB 是 HTML 解析输入限制，不是网络响应内存限制——loadURL 会先把整包
//   响应下载+解码再量尺寸（HTTP 层响应体上限是后续项，不在本轮）；
// - 解析后走树计数：节点 > 100,000 抛 HTMLDocumentTooLargeError，深度 > 512 抛 HTMLDocumentTooDeepError；
//   这是解析后护栏（防大树被保留/反复扫），解析本身的代价由 5MB 输入上限兜底——
//   解析器层限额（parse5 treeAdapter 计数）是后续改进，不要把它说成解析中护栏；
// - loadURL 非 2xx 时 reject，错误名 HTMLHTTPError，带 .status 与 .finalUrl（不静默返回空文档）；
//   loadURL 的 timeout 沿用 Widget.http.get 的语义：钳制在 3–25 秒之间，默认 12 秒，
//   传 timeout: 500 不保证 500ms 超时；
// - 字符集：沿用 Widget.http.get 的解码（UTF-8，失败时按 UTF-8 宽容解码），
//   GBK 等非 UTF-8 页面目前会乱码（与现有模块 http.get + 正则的现状一致），以后 http 层做字符集嗅探时这里自动受益；
// - prop('href')/prop('src') 的相对地址解析：<a>/<link> 的 href 与
//   img/iframe/audio/video/source 的 src 会对 baseURI 解析（cheerio 原生行为，
//   依赖全局 URL；裸 JSContext 里没有 URL，宿主在建桥时就装了一个最小 URL 构造器
//   （模块在第一次调 html.load 之前 new URL 也能用）；searchParams 是活的，
//   set/append/delete 后会同步回 URL 的 search 与 href）；
//   别的标签的 prop 返回原始值。attr() 永远返回原始值。
// - baseURI：loadURL 默认取响应的 finalUrl；load 的 options.baseURI 可显式传。
// - 解析跑在主线程 JSContext 里（与现有模块一致），典型列表/详情页（50KB–800KB）约 20–50ms。
(function (g) {
    'use strict';

    var HARD_MAX_HTML_BYTES = 5 * 1024 * 1024;
    var MAX_DOM_NODES = 100000;
    var MAX_DOM_DEPTH = 512;

    // 解析输入上限：硬上限 5MB，只许调低不许调高（模块是外部 JS，不能拿它给的配置当资源策略）。
    // 未传或 null → 默认 5MB；maxBytes 与 maxHTMLBytes 两个别名都认（maxBytes 优先）；
    // 无效值（非数/NaN/Infinity/<1）抛 TypeError，不悄悄回默认——注意 0.5 取整是 0，
    // 所以下限按 >= 1 判，不能只判 > 0。
    function resolveMaxBytes(options) {
        if (!options) { return HARD_MAX_HTML_BYTES; }
        var raw = options.maxBytes;
        if (raw === undefined || raw === null) { raw = options.maxHTMLBytes; }
        if (raw === undefined || raw === null) { return HARD_MAX_HTML_BYTES; }
        if (typeof raw !== 'number' || !isFinite(raw) || raw < 1) {
            throw new TypeError('Widget.html.load: maxBytes must be a finite number >= 1');
        }
        return Math.min(Math.floor(raw), HARD_MAX_HTML_BYTES);
    }

    // UTF-8 字节数。不用 TextEncoder：它在 JSC 里是否可用不确定，手动算是确定的。
    function utf8ByteLength(s) {
        var n = 0;
        for (var i = 0; i < s.length; i++) {
            var c = s.charCodeAt(i);
            if (c < 0x80) { n += 1; }
            else if (c < 0x800) { n += 2; }
            else if (c >= 0xD800 && c <= 0xDBFF && i + 1 < s.length) {
                var d = s.charCodeAt(i + 1);
                if (d >= 0xDC00 && d <= 0xDFFF) { n += 4; i++; }
                else { n += 3; }
            } else { n += 3; }
        }
        return n;
    }

    // 迭代式走树（不用递归：深树会爆 JS 栈），数节点数与最大深度，超限就抛。
    // 在解析完成后跑：它防的是超大树被后续选择器反复扫的代价，解析本身的代价由 5MB 输入上限兜底。
    function checkTree(root) {
        var nodes = 0;
        var maxDepth = 0;
        var stack = [{ n: root, d: 0 }];
        while (stack.length > 0) {
            var cur = stack.pop();
            nodes++;
            if (nodes > MAX_DOM_NODES) {
                var e1 = new Error('Widget.html.load: DOM node count exceeds ' + MAX_DOM_NODES);
                e1.name = 'HTMLDocumentTooLargeError';
                throw e1;
            }
            if (cur.d > maxDepth) { maxDepth = cur.d; }
            if (maxDepth > MAX_DOM_DEPTH) {
                var e2 = new Error('Widget.html.load: DOM depth exceeds ' + MAX_DOM_DEPTH);
                e2.name = 'HTMLDocumentTooDeepError';
                throw e2;
            }
            var kids = cur.n.children;
            if (kids && kids.length) {
                for (var i = kids.length - 1; i >= 0; i--) {
                    stack.push({ n: kids[i], d: cur.d + 1 });
                }
            }
        }
    }

    // cheerio 选项：只透传已知的三个，别的键不往下传（防模块误传 maxBytes 之类被 cheerio 误解）。
    function cheerioOptionsFrom(options) {
        var out = {};
        if (options) {
            if (typeof options.baseURI === 'string') { out.baseURI = options.baseURI; }
            if (typeof options.decodeEntities === 'boolean') { out.decodeEntities = options.decodeEntities; }
            if (typeof options.xml === 'boolean') { out.xml = options.xml; }
        }
        return out;
    }

    function loadImpl(html, options) {
        if (typeof html !== 'string') {
            throw new TypeError('Widget.html.load: html must be a string');
        }
        options = options || {};
        var maxBytes = resolveMaxBytes(options);
        if (utf8ByteLength(html) > maxBytes) {
            var e = new Error('Widget.html.load: HTML document exceeds ' + maxBytes + ' bytes');
            e.name = 'HTMLDocumentTooLargeError';
            throw e;
        }
        if (typeof g.__fwEnsureCheerio !== 'function' || !g.__fwEnsureCheerio()) {
            throw new Error('Widget.html: cheerio runtime is unavailable');
        }
        if (typeof g.__fwCheerioLoad !== 'function') {
            throw new Error('Widget.html: cheerio runtime is unavailable');
        }
        var $ = g.__fwCheerioLoad(html, cheerioOptionsFrom(options));
        var root = $.root();
        if (root && root.length && root[0]) { checkTree(root[0]); }
        return $;
    }

    function loadURLImpl(url, options) {
        options = options || {};
        // url 必须是字符串；null/undefined 不能 String() 成 "null"/"undefined" 再发出去。
        if (url == null) {
            return Promise.reject(new TypeError('Widget.html.loadURL: url must be a string'));
        }
        url = String(url);
        var httpOptions = {};
        if (options.headers) { httpOptions.headers = options.headers; }
        if (options.timeout !== undefined && options.timeout !== null) { httpOptions.timeout = options.timeout; }
        var baseURIOverride = options.baseURI;
        // maxBytes 先验（无效就直接拒绝，不发请求；合法值由 loadImpl 里再钳制一次）。
        try { resolveMaxBytes(options); } catch (e) { return Promise.reject(e); }
        return g.Widget.http.get(url, httpOptions).then(function (res) {
            // res 可能为空、status 可能不是数字：非数字/NaN/Infinity/小数按 0 算，非 2xx（含 0）
            // 按原约定抛 HTMLHTTPError（带 .status 与 .finalUrl），不静默返回空文档。
            // typeof NaN 是 number，但 NaN 和任何数比较都是 false，会漏过非 2xx 检查，
            // 所以要同时要求 isFinite 与整数。
            var status = (res && typeof res.status === 'number' && isFinite(res.status) && Math.floor(res.status) === res.status) ? res.status : 0;
            var finalUrl = (res && res.finalUrl) || url;
            if (status < 200 || status >= 300) {
                var e = new Error('Widget.html.loadURL: HTTP ' + status + ' for ' + url);
                e.name = 'HTMLHTTPError';
                e.status = status;
                e.finalUrl = finalUrl;
                throw e;
            }
            // data 可能是宿主解析过的 JSON 对象（见 WidgetHost.httpDataObject）：
            // 字符串原样用，对象转 JSON 文本（与注入的 fetch 的 text() 同款处理）。
            var html = (res.data == null) ? '' : (typeof res.data === 'string' ? res.data : JSON.stringify(res.data));
            var loadOptions = {};
            if (options.maxBytes !== undefined && options.maxBytes !== null) { loadOptions.maxBytes = options.maxBytes; }
            if (options.maxHTMLBytes !== undefined && options.maxHTMLBytes !== null) { loadOptions.maxHTMLBytes = options.maxHTMLBytes; }
            if (typeof options.xml === 'boolean') { loadOptions.xml = options.xml; }
            if (typeof options.decodeEntities === 'boolean') { loadOptions.decodeEntities = options.decodeEntities; }
            loadOptions.baseURI = (typeof baseURIOverride === 'string' && baseURIOverride)
                ? baseURIOverride
                : finalUrl;
            return loadImpl(html, loadOptions);
        });
    }

    g.Widget.html = {
        load: loadImpl,
        loadURL: loadURLImpl
    };
})(typeof globalThis !== 'undefined' ? globalThis : this);
